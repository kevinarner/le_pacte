#!/bin/bash
# Suites SQL métier sur une base où l'échéance H-3 vient du paquet D-027-03.
. /home/user/le_pacte/qa/lib/commun.sh
S=/tmp/claude-0/-home-user-le-pacte/05ea20a0-4299-572a-9d63-2d5619bb913e/scratchpad
D=/home/user/le_pacte/supabase/changements/D-027-03
B=swend_d027_03_suites
$S/base_avant_d03.sh $B >/dev/null || exit 1
for f in sauvegarde appliquer verifier; do qa_psql $B -v ON_ERROR_STOP=1 -q -f $D/$f.sql >/dev/null 2>&1 || { echo "ECHEC $f"; exit 1; }; done
echo "# D-027-03 — suites SQL métier sur une base où l'échéance H-3 vient du paquet"
ok=0; ko=0
for f in /home/user/le_pacte/qa/metier/sql/*.sql; do
  if ! qa_psql $B -v ON_ERROR_STOP=1 -f "$f" > $S/d03_$(basename $f).log 2>&1; then echo "$(basename $f) : ARRÊTÉE"; ko=$((ko+1)); continue; fi
  r=$(qa_psql $B -tA -F' ' -c "select count(*) filter (where ok), count(*) filter (where not ok) from test_resultats")
  echo "$(basename $f) : ${r% *} réussies, ${r#* } échecs"
  ok=$((ok + ${r% *})); ko=$((ko + ${r#* }))
done
echo "TOTAL suites SQL sur base issue du paquet : $ok réussies, $ko échecs"
